# Классификация наблюдаемого сбоя одной попытки запуска Python-приложения
ЕСЛИ Ошибка_запуска=модуль_не_найден ТО Сбой_импорта=да
ЕСЛИ Ошибка_запуска=ошибка_базы ТО Сбой_базы=да
ЕСЛИ Ошибка_запуска=синтаксис ТО Сбой_разбора_кода=да
ЕСЛИ Ошибка_запуска=настройки ТО Сбой_конфигурации=да
ЕСЛИ Ошибка_запуска=привязка_порта ТО Сбой_прослушивания=да

# Один симптом ModuleNotFoundError — две разные причины
ЕСЛИ Окружение_верное=да И Пакет_установлен_в_нужной_среде=нет ТО Дефект_зависимости=да
ЕСЛИ Окружение_верное=нет И Пакет_установлен_в_нужной_среде=да ТО Дефект_окружения=да
ЕСЛИ Сбой_импорта=да И Дефект_зависимости=да ТО Причина=отсутствует_зависимость
ЕСЛИ Сбой_импорта=да И Дефект_окружения=да ТО Причина=неверное_окружение_Python

# Сообщение сверяется с независимой проверкой проекта
ЕСЛИ Синтаксис_проверен_без_ошибок=нет ТО Дефект_кода=да
ЕСЛИ Сбой_разбора_кода=да И Дефект_кода=да ТО Причина=синтаксическая_ошибка
ЕСЛИ Обязательные_настройки_заданы=нет ТО Дефект_конфигурации=да
ЕСЛИ Сбой_конфигурации=да И Дефект_конфигурации=да ТО Причина=отсутствует_обязательная_настройка

# Альтернативные признаки невозможности установить соединение
ЕСЛИ Сообщение_драйвера=connection_refused ТО Соединение_с_базой=не_установлено
ЕСЛИ Сообщение_драйвера=timeout ТО Соединение_с_базой=не_установлено
ЕСЛИ Сообщение_драйвера=host_not_resolved ТО Соединение_с_базой=не_установлено

# Контрольный запуск меняет только один параметр, остальные оставляет прежними
ЕСЛИ Адрес_базы_верный=нет И После_исправления_адреса_соединение_есть=да ТО Дефект_адреса=да
ЕСЛИ Порт_базы_верный=нет И После_исправления_порта_соединение_есть=да ТО Дефект_порта_базы=да
ЕСЛИ Адрес_базы_верный=да И Порт_базы_верный=да ТО Параметры_маршрута=проверены
ЕСЛИ Параметры_маршрута=проверены И База_запущена=нет ТО Служба_базы=остановлена
ЕСЛИ Параметры_маршрута=проверены И База_доступна_локально=да И База_доступна_из_приложения=нет ТО Сетевой_маршрут=недоступен

# Общий симптом — разные подтверждающие проверки
ЕСЛИ Сбой_базы=да И Соединение_с_базой=не_установлено И Дефект_адреса=да ТО Причина=неверный_адрес_базы
ЕСЛИ Сбой_базы=да И Соединение_с_базой=не_установлено И Дефект_порта_базы=да ТО Причина=неверный_порт_базы
ЕСЛИ Сбой_базы=да И Сообщение_драйвера=connection_refused И Служба_базы=остановлена ТО Причина=база_не_запущена
ЕСЛИ Сбой_базы=да И Сообщение_драйвера=timeout И Сетевой_маршрут=недоступен ТО Причина=недоступен_сетевой_маршрут_к_базе

# Авторизация отличается от недоступности сервера; допустим текст или SQLSTATE
ЕСЛИ Сообщение_драйвера=password_authentication_failed ТО Сбой_авторизации=да
ЕСЛИ Сообщение_драйвера=28P01 ТО Сбой_авторизации=да
ЕСЛИ Сбой_базы=да И Сбой_авторизации=да И Параметры_маршрута=проверены И Проверочные_учётные_данные_приняты=нет ТО Причина=неверные_учётные_данные_базы
ЕСЛИ Сообщение_драйвера=database_does_not_exist ТО Сбой_выбора_базы=да
ЕСЛИ Сообщение_драйвера=3D000 ТО Сбой_выбора_базы=да
ЕСЛИ Сбой_базы=да И Сбой_выбора_базы=да И Параметры_маршрута=проверены И База_существует=нет ТО Причина=указанная_база_отсутствует

# Проверяется владелец занятого порта приложения
ЕСЛИ Порт_приложения_занят=да И Порт_занят_другим_процессом=да ТО Конфликт_порта=да
ЕСЛИ Сбой_прослушивания=да И Конфликт_порта=да ТО Причина=порт_приложения_занят

# Следующий шаг выводится из доказанной причины
ЕСЛИ Причина=отсутствует_зависимость ТО Рекомендация=установить_зависимости_проекта_в_нужное_окружение
ЕСЛИ Причина=неверное_окружение_Python ТО Рекомендация=выбрать_интерпретатор_нужного_окружения
ЕСЛИ Причина=синтаксическая_ошибка ТО Рекомендация=исправить_код_в_месте_ошибки_и_повторить_проверку_синтаксиса
ЕСЛИ Причина=отсутствует_обязательная_настройка ТО Рекомендация=задать_отсутствующий_параметр_по_настройкам_проекта
ЕСЛИ Причина=неверный_адрес_базы ТО Рекомендация=исправить_адрес_базы_в_конфигурации_приложения
ЕСЛИ Причина=неверный_порт_базы ТО Рекомендация=исправить_порт_базы_в_конфигурации_приложения
ЕСЛИ Причина=база_не_запущена ТО Рекомендация=запустить_нужную_службу_базы_и_повторить_подключение
ЕСЛИ Причина=недоступен_сетевой_маршрут_к_базе ТО Рекомендация=проверить_маршрут_и_правила_сетевого_доступа
ЕСЛИ Причина=неверные_учётные_данные_базы ТО Рекомендация=сверить_пользователя_и_способ_авторизации_с_настройками_базы
ЕСЛИ Причина=указанная_база_отсутствует ТО Рекомендация=сверить_имя_базы_и_подготовить_её_по_инструкции_проекта
ЕСЛИ Причина=порт_приложения_занят ТО Рекомендация=выбрать_свободный_порт_или_разобраться_с_процессом_владельцем
